#Create a project folder structure

#Giving permissions to execute the file
chmod 777 create-service

#Reading the content script of the file
cat create-service

#Using shebang operator
#!/bin/bash

#Creating folders and files required for the script
mkdir -p service/controllers
mkdir -p service/models
mkdir -p service/routes
mkdir -p service/middleware
printf '%s\n' 'console.log("Server started");' > service/server.js
printf '%s\n' 'API Service' > service/README.md
